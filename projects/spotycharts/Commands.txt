# Step 1: Create directories in HDFS
hdfs dfs -mkdir -p /input

# Step 2: Upload dataset to HDFS
hdfs dfs -put charts_trimmed.csv /input

# Step 3: Compile and package the first MapReduce job
javac -classpath $(hadoop classpath) -d . TotalStreamsByArtist.java
jar -cvf TotalStreamsByArtist.jar TotalStreamsByArtist*.class

# Step 4: Run the first MapReduce job
hadoop jar TotalStreamsByArtist.jar TotalStreamsByArtist /input /output_total_streams_by_artist

# Step 5: View the output of the first job
hdfs dfs -cat /output_total_streams_by_artist/part-r-00000

# Step 6: Compile and package the second MapReduce job
javac -classpath $(hadoop classpath) -d . TopRegionsByStreams.java
jar -cvf TopRegionsByStreams.jar TopRegionsByStreams*.class

# Step 7: Run the second MapReduce job
hdfs dfs -rm -r /output_top_regions_by_streams
hadoop jar TopRegionsByStreams.jar TopRegionsByStreams /input /output_top_regions_by_streams

# Step 8: View the output of the second job
hdfs dfs -cat /output_top_regions_by_streams/part-r-00000

# Step 9: Compile and package the third MapReduce job
javac -classpath $(hadoop classpath) -d . AverageStreamsByTrend.java
jar -cvf AverageStreamsByTrend.jar AverageStreamsByTrend*.class

# Step 10: Run the third MapReduce job
hdfs dfs -rm -r /output_avg_streams_by_trend
hadoop jar AverageStreamsByTrend.jar AverageStreamsByTrend /input /output_avg_streams_by_trend

# Step 11: View the output of the third job
hdfs dfs -cat /output_avg_streams_by_trend/part-r-00000
